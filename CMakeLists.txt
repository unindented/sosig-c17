cmake_minimum_required(VERSION 3.25...4.0)

# Set this target before `project()` because compiler checks use it. Builders can raise the value.
set(CMAKE_OSX_DEPLOYMENT_TARGET
    "13.0"
    CACHE STRING "Minimum macOS version the build targets"
)

project(
  sosig
  VERSION 0.1.1
  DESCRIPTION "A small static site generator written in C17"
  HOMEPAGE_URL "https://github.com/unindented/sosig-c17"
  LANGUAGES C
)

if(PROJECT_IS_TOP_LEVEL AND "${PROJECT_SOURCE_DIR}" STREQUAL "${PROJECT_BINARY_DIR}")
  message(FATAL_ERROR "in-source builds are unsupported; use a preset or '-B build/<name>'")
endif()

if(NOT DEFINED SOSIG_BUILD_TESTING)
  set(SOSIG_BUILD_TESTING "${PROJECT_IS_TOP_LEVEL}")
endif()

if(SOSIG_BUILD_TESTING)
  if(PROJECT_IS_TOP_LEVEL)
    enable_testing()
  else()
    message(STATUS "sosig tests are enabled as a child; the parent must call 'enable_testing()' at its top level")
  endif()
endif()

if(NOT DEFINED CMAKE_C_STANDARD)
  set(CMAKE_C_STANDARD 17)
endif()
if(NOT DEFINED CMAKE_C_STANDARD_REQUIRED)
  set(CMAKE_C_STANDARD_REQUIRED TRUE)
endif()
if(NOT DEFINED CMAKE_C_EXTENSIONS)
  set(CMAKE_C_EXTENSIONS FALSE)
endif()

set(CMAKE_LINK_LIBRARIES_ONLY_TARGETS TRUE)
set(CMAKE_EXPORT_COMPILE_COMMANDS TRUE)

if(PROJECT_IS_TOP_LEVEL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
endif()

list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")

set(THREADS_PREFER_PTHREAD_FLAG TRUE)
find_package(Threads REQUIRED)

include(sosig_build_profiles)

if(PROJECT_IS_TOP_LEVEL)
  include(sosig_quality)
endif()
if(SOSIG_BUILD_TESTING)
  include(sosig_testing)
endif()

add_subdirectory(vendor)
add_subdirectory(src)

if(SOSIG_BUILD_TESTING)
  add_subdirectory(tests)
endif()

if(PROJECT_IS_TOP_LEVEL)
  include(sosig_packaging)
endif()
